cmake_minimum_required(VERSION 3.20)
project(warbeast_body_m94 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(m94_body_core_smoke body_resolver_smoke.cpp)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(m94_body_core_smoke PRIVATE -Wall -Wextra -Werror -UNDEBUG -fno-fast-math -ffp-contract=off)
endif()
enable_testing()
add_test(NAME m94_body_core_smoke COMMAND m94_body_core_smoke)

option(M94_BUILD_GDEXTENSION "Build only with explicit reviewed fixed godot-cpp inputs" OFF)
if(M94_BUILD_GDEXTENSION)
  if(NOT M94_GODOT_CPP_DIR OR NOT EXISTS "${M94_GODOT_CPP_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "Supply reviewed godot-4.5-stable source at M94_GODOT_CPP_DIR; no dependency is downloaded")
  endif()
  file(SHA256 "${M94_GODOT_CPP_DIR}/gdextension/extension_api.json" api_sha)
  file(SHA256 "${M94_GODOT_CPP_DIR}/gdextension/gdextension_interface.h" interface_sha)
  if(NOT api_sha STREQUAL "23d807e3f914f7a91b152a8a7b03638d4853f8e642b79f10ce85a43b44340bfd"
      OR NOT interface_sha STREQUAL "a40ac4fca0f526910bd0e6afc6da6c169f50801c84d4e29c4ce2891cadc7b550")
    message(FATAL_ERROR "godot-cpp API/interface differ from VENDOR_LOCK.json")
  endif()
  add_library(warbeast_body_m94 SHARED body_resolver_binding.cpp)
  if(M94_GODOT_CPP_ARCHIVE)
    if(NOT EXISTS "${M94_GODOT_CPP_ARCHIVE}" OR NOT EXISTS "${M94_GODOT_CPP_GENERATED_DIR}/include")
      message(FATAL_ERROR "Explicit archive mode also requires matching M94_GODOT_CPP_GENERATED_DIR")
    endif()
    target_include_directories(warbeast_body_m94 PRIVATE "${M94_GODOT_CPP_DIR}/include"
      "${M94_GODOT_CPP_GENERATED_DIR}/include" "${M94_GODOT_CPP_DIR}/gdextension")
    target_link_libraries(warbeast_body_m94 PRIVATE "${M94_GODOT_CPP_ARCHIVE}")
  else()
    set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/godotcpp-profile.json" CACHE FILEPATH "Reviewed class profile")
    set(GODOTCPP_GDEXTENSION_DIR "${M94_GODOT_CPP_DIR}/gdextension" CACHE PATH "Locked API")
    set(GODOTCPP_TARGET template_release CACHE STRING "Matching supplied binary target")
    set(GODOTCPP_GENERATE_TEMPLATE_GET_NODE OFF CACHE BOOL "No node template")
    set(GODOTCPP_USE_HOT_RELOAD OFF CACHE BOOL "No reload")
    set(GODOTCPP_ENABLE_TESTING OFF CACHE BOOL "No upstream project tests")
    add_subdirectory("${M94_GODOT_CPP_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/godot-cpp")
    target_link_libraries(warbeast_body_m94 PRIVATE godot::cpp)
  endif()
  target_compile_definitions(warbeast_body_m94 PRIVATE GDEXTENSION THREADS_ENABLED)
  if(CMAKE_SYSTEM_NAME STREQUAL "Darwin")
    target_compile_definitions(warbeast_body_m94 PRIVATE MACOS_ENABLED UNIX_ENABLED)
  endif()
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(warbeast_body_m94 PRIVATE -fno-fast-math -ffp-contract=off -fno-exceptions)
  endif()
  set_target_properties(warbeast_body_m94 PROPERTIES
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/package/bin"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/package/bin"
    PREFIX "lib" OUTPUT_NAME "warbeast_body_m94")
  configure_file(warbeast_body_m94.gdextension "${CMAKE_CURRENT_BINARY_DIR}/package/warbeast_body_m94.gdextension" COPYONLY)
endif()
